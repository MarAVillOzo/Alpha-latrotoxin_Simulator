    # ============================================================
    # MOTOR BIOFISICO FENOMENOLOGICO
    # ============================================================

    def step(
        self,
        dt
    ):

        f = max(
            0,
            self.stim.get()
        )

        ce = max(
            0.01,
            self.ca_ext.get()
        )

        cav = max(
            0,
            self.cav.get()
        )

        l = float(
            np.clip(
                self.ltx.get(),
                0,
                1
            )
        )

        ex = max(
            0.05,
            self.exo.get()
        )

        cap = max(
            1,
            self.pool.get()
        )

        tau = max(
            0.1,
            self.recycle.get()
        )

        # --------------------------------------------------------
        # Ca2+
        # --------------------------------------------------------
        #
        # Entrada evocada:
        # canales de calcio dependientes de actividad
        #
        # Entrada persistente:
        # efecto tipo poro de alpha-LTX virtual
        #
        # Aclaramiento:
        # retorno hacia concentracion basal
        #

        influx = (
            0.52 *
            cav *
            (ce / 2) *
            (f / 20)
            +
            2.20 *
            l *
            (ce / 2)
        )

        self.ca = max(
            0,
            self.ca
            +
            (
                influx
                -
                (self.ca - 0.08) / 0.24
            )
            *
            dt
        )

        # --------------------------------------------------------
        # LIBERACION VESICULAR
        # --------------------------------------------------------

        z = float(
            np.clip(
                -8 *
                ex *
                (self.ca - 0.22),
                -60,
                60
            )
        )

        self.pr = (
            1 /
            (
                1 +
                math.exp(z)
            )
        )

        wanted = (
            f *
            self.pr
            +
            24 *
            l *
            ex
        )

        self.release = min(
            max(
                0,
                wanted
            ),
            self.ves /
            max(
                dt,
                1e-9
            )
        )

        # --------------------------------------------------------
        # RECICLAJE DE VESICULAS
        # --------------------------------------------------------

        self.ves = float(
            np.clip(
                self.ves
                +
                (
                    (
                        cap -
                        self.ves
                    )
                    /
                    tau
                    -
                    self.release
                )
                *
                dt,
                0,
                cap
            )
        )

        self.total_release += (
            self.release *
            dt
        )

        # --------------------------------------------------------
        # NEUROTRANSMISOR
        # --------------------------------------------------------

        self.nt = max(
            0,
            self.nt
            +
            (
                0.12 *
                self.release
                -
                self.nt /
                0.12
            )
            *
            dt
        )

        # --------------------------------------------------------
        # RESPUESTA POSTSINAPTICA
        # --------------------------------------------------------

        act = (
            self.nt /
            (
                0.28 +
                self.nt +
                1e-12
            )
        )

        target = (
            9 *
            act
        )

        self.epsp += (
            (
                target -
                self.epsp
            )
            /
            0.18
        ) * dt

        # --------------------------------------------------------
        # INDICE DE INESTABILIDAD
        # --------------------------------------------------------
        #
        # NO es una escala de toxicidad real.
        # Sirve para comparar estados dentro del modelo.
        #

        a = max(
            0,
            (
                self.ca -
                0.45
            )
            /
            0.75
        )

        b = max(
            0,
            (
                self.nt -
                0.35
            )
            /
            1.2
        )

        d = max(
            0,
            1 -
            self.ves /
            cap
        )

        self.instability = float(
            np.clip(
                55*a +
                25*b +
                35*d,
                0,
                100
            )
        )

        # --------------------------------------------------------
        # TIEMPO
        # --------------------------------------------------------

        self.t += dt

        valores = [
            ("t", self.t),
            ("ca", self.ca),
            ("nt", self.nt),
            ("epsp", self.epsp),
            ("release", self.release),
            ("ves", self.ves),
            ("ltx", l),
            (
                "instability",
                self.instability
            )
        ]

        for k, v in valores:
            self.H[k].append(v)

        # Maximo de historia
        if len(
            self.H["t"]
        ) > 12000:

            for k in self.H:

                self.H[k] = (
                    self.H[k][-12000:]
                )

    # ============================================================
    # LOOP PRINCIPAL
    # ============================================================

    def loop(self):

        if self.running:

            dur = (
                0.05 *
                max(
                    0.05,
                    self.speed.get()
                )
            )

            n = max(
                1,
                math.ceil(
                    dur /
                    self.dt
                )
            )

            for _ in range(n):

                self.step(
                    dur /
                    n
                )

            self.update_all()

        self.root.after(
            50,
            self.loop
        )

    # ============================================================
    # ACTUALIZACION VISUAL
    # ============================================================

    def update_all(self):

        cap = max(
            1,
            self.pool.get()
        )

        frac = np.clip(
            self.ves /
            cap,
            0,
            1
        )

        n = int(
            frac *
            len(self.vx)
        )

        # Vesiculas
        self.sves._offsets3d = (
            self.vx[:n],
            self.vy[:n],
            self.vz[:n]
        )

        # Ca2+
        nca = int(
            np.clip(
                18 +
                self.ca *
                90,
                18,
                140
            )
        )

        cx = self.rng.uniform(
            -0.45,
            0.45,
            nca
        )

        cy = self.rng.uniform(
            -0.45,
            0.45,
            nca
        )

        cz = self.rng.uniform(
            0.002,
            0.07,
            nca
        )

        self.sca._offsets3d = (
            cx,
            cy,
            cz
        )

        # Neurotransmisor
        nnt = int(
            np.clip(
                self.nt *
                260,
                0,
                180
            )
        )

        nx = np.clip(
            self.rng.normal(
                0,
                0.23,
                nnt
            ),
            -0.48,
            0.48
        )

        ny = np.clip(
            self.rng.normal(
                0,
                0.23,
                nnt
            ),
            -0.48,
            0.48
        )

        nz = self.rng.uniform(
            -0.019,
            -0.001,
            nnt
        )

        self.snt._offsets3d = (
            nx,
            ny,
            nz
        )

        # alpha-LTX
        nl = int(
            self.ltx.get() *
            18
        )

        if nl:

            a = np.linspace(
                0,
                2*np.pi,
                nl,
                endpoint=False
            )

            lx = (
                0.32 *
                np.cos(a)
            )

            ly = (
                0.32 *
                np.sin(a)
            )

            lz = np.full(
                nl,
                0.003
            )

        else:

            lx = np.array([])
            ly = np.array([])
            lz = np.array([])

        self.sltx._offsets3d = (
            lx,
            ly,
            lz
        )

        self.info.set_text(
            f"t={self.t:.2f}s\n"
            f"alpha-LTX={self.ltx.get():.2f}\n"
            f"Ca2+={self.ca:.3f} uM\n"
            f"release={self.release:.1f} ves/s"
        )

        # --------------------------------------------------------
        # GRAFICAS TEMPORALES
        # --------------------------------------------------------

        if self.H["t"]:

            t = np.array(
                self.H["t"]
            )

            ca = np.array(
                self.H["ca"]
            )

            nt = np.array(
                self.H["nt"]
            )

            rr = np.array(
                self.H["release"]
            )

            self.lca.set_data(
                t,
                ca
            )

            self.lnt.set_data(
                t,
                nt
            )

            self.lrr.set_data(
                t,
                rr
            )

            x1 = max(
                30,
                self.t
            )

            x0 = max(
                0,
                x1 -
                30
            )

            mask = (
                t >= x0
            )

            self.ax.set_xlim(
                x0,
                x1
            )

            ymax = max(
                np.max(
                    ca[mask]
                ),
                np.max(
                    nt[mask]
                )
            )

            self.ax.set_ylim(
                0,
                max(
                    0.5,
                    float(ymax) *
                    1.15
                )
            )

            self.axr.set_ylim(
                0,
                max(
                    10,
                    float(
                        np.max(
                            rr[mask]
                        )
                    )
                    *
                    1.2
                )
            )

        # --------------------------------------------------------
        # METRICAS
        # --------------------------------------------------------

        self.metrics["ca"].set(
            f"{self.ca:.3f} uM"
        )

        self.metrics["pr"].set(
            f"{self.pr:.3f}"
        )

        self.metrics["rr"].set(
            f"{self.release:.2f} ves/s"
        )

        self.metrics["nt"].set(
            f"{self.nt:.3f} uM*"
        )

        self.metrics["epsp"].set(
            f"{self.epsp:.3f} mV*"
        )

        self.metrics["ves"].set(
            f"{self.ves:.0f}/{cap:.0f}"
        )

        self.metrics["total"].set(
            f"{self.total_release:.1f}"
        )

        self.metrics["idx"].set(
            f"{self.instability:.1f}/100*"
        )

        self.canvas.draw_idle()

    # CONTROLES
    # ============================================================

    def toggle(self):

        self.running = (
            not
            self.running
        )

        self.runbtn.config(
            text=(
                "Pausar"
                if self.running
                else
                "Iniciar"
            )
        )

        self.log(
            "Simulacion iniciada."
            if self.running
            else
            "Simulacion pausada."
        )

    def inject(self):

        old = self.ltx.get()

        self.ltx.set(
            min(
                1,
                old +
                0.25
            )
        )

        self.log(
            f"alpha-LTX virtual: "
            f"{old:.2f} -> "
            f"{self.ltx.get():.2f}"
        )

        self.update_all()

    def washout(self):

        self.ltx.set(
            0
        )

        self.log(
            "Washout virtual aplicado."
        )

        self.update_all()

    def reset(self):

        if hasattr(
            self,
            "runbtn"
        ):

            self.runbtn.config(
                text="Iniciar"
            )

        self.t = 0
        self.ca = 0.08
        self.nt = 0
        self.epsp = 0
        self.release = 0
        self.pr = 0
        self.total_release = 0
        self.instability = 0
        self.ves = self.pool.get()

        for k in self.H:

            self.H[k].clear()

        if hasattr(
            self,
            "canvas"
        ):

            self.update_all()

    # ============================================================
    # EXPORTACION
    # ============================================================

    def export_csv(self):

        if not self.H["t"]:

            messagebox.showinfo(
                "MainLab",
                "Aun no hay datos "
                "para exportar."
            )

            return

        p = filedialog.asksaveasfilename(
            defaultextension=".csv",
            filetypes=[
                (
                    "CSV",
                    "*.csv"
                )
            ],
            initialfile="mainlab_experiment.csv"
        )

        if not p:
            return

        with open(
            p,
            "w",
            newline="",
            encoding="utf-8"
        ) as f:

            w = csv.writer(f)

            w.writerow(
                [
                    "time_s",
                    "ca_uM",
                    "nt_model_uM",
                    "epsp_model_mV",
                    "release_ves_s",
                    "vesicles_ready",
                    "alpha_ltx_virtual",
                    "instability_model"
                ]
            )

            w.writerows(
                zip(
                    *[
                        self.H[k]
                        for k in [
                            "t",
                            "ca",
                            "nt",
                            "epsp",
                            "release",
                            "ves",
                            "ltx",
                            "instability"
                        ]
                    ]
                )
            )

        self.log(
            "CSV exportado: " +
            p
        )

    # ============================================================
    # LOG
    # ============================================================

    def log(
        self,
        msg
    ):

        if not hasattr(
            self,
            "status"
        ):

            return

        self.status.insert(
            "end",
            f"[{self.t:7.2f}s] "
            f"{msg}\n"
        )

        self.status.see(
            "end"
        )
